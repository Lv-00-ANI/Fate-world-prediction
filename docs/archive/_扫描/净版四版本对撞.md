          "CRLF=", b.count(b'\r\n'), "LF=", b.count(b'\n'), "BOM=", b[:3] == b'\xef\xbb\xbf',
          "lines=", len(L[k]), "chars=", len(TEXT[k]))

# 1310 vs 766：逐行 diff，分别累计删除/新增字节
def bytediff(a, b):                      # b 相对 a
    sm = difflib.SequenceMatcher(None, L[a], L[b], autojunk=False)
    d = i = n = 0
    for tag, i1, i2, j1, j2 in sm.get_opcodes():
        if tag == 'equal':
            continue
        n += 1
        d += sum(len(L[a][x].encode()) + 1 for x in range(i1, i2))
        i += sum(len(L[b][y].encode()) + 1 for y in range(j1, j2))
    return n, d, i
print("1310 vs 766:", bytediff(766, 1310), "actual byte diff =", len(raw[1310]) - len(raw[766]))
print("766 vs 545 :", bytediff(545, 766), "actual byte diff =", len(raw[766]) - len(raw[545]))
